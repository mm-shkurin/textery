# Runtime dependencies only -- what the production image installs.
#
# Split from the development tooling on 2026-08-07. They were one file, and
# infra/docker/backend.Dockerfile installs this file, so pytest, pytest-cov,
# pytest-mock, ruff, mypy and a stubs package all shipped into the production
# image: dead weight, extra attack surface, and a `pip-audit` result that could
# not answer "is what we deploy vulnerable" because it was auditing the linter
# too. Development tooling now lives in requirements-dev.txt, which installs
# this file first.
#
# Every version is pinned exactly. That is what makes the build reproducible --
# and also what makes a known vulnerability sit here until the `audit` CI job
# reports it, which is why that job is blocking.

fastapi==0.139.0
uvicorn==0.34.0
httpx==0.28.1
sqlalchemy==2.0.49
asyncpg==0.31.0
bcrypt==5.0.0
PyJWT==2.13.0
# HTML sanitizer for manual-document content (story 5). Rust/ammonia binding.
# bleach is archived/EOL since Jan 2023 and its own README points here.
nh3==0.3.6
# HTML -> PDF renderer for document export (story 17). Needs native Pango/cairo
# libs, installed in the backend image; cannot import on a host without them, so
# its adapter tests run in-container and on CI, never on a bare dev host.
# 69.0 rather than the 63.1 this shipped on: 63.1 carried PYSEC-2026-2034 and
# PYSEC-2026-3412, and 68.0 -- the fix version the first advisory names -- still
# carries the second. The renderer touches only `HTML(string=, url_fetcher=)`
# and `write_pdf(stylesheets=[CSS(string=)])`, unchanged across those releases.
weasyprint==69.0
# HTML -> DOCX renderer for document export (story 17, Sc 2.2). Both pure-Python
# (no native OS libs), so unlike weasyprint they need no apt packages in CI or
# the backend image -- adding them to this file is enough. htmldocx parses the
# already-sanitized HTML into a python-docx Document; python-docx exposes
# core_properties so the renderer can redact author/time metadata.
python-docx==1.1.2
htmldocx==0.0.6
# Markdown -> HTML for the generation-to-document conversion (story 18). The LLM
# answers in markdown; the editor and every export downstream speak HTML, so the
# conversion is what makes a generated document editable at all. Pure-Python, no
# native libs. Its output is NOT trusted -- it goes through Nh3HtmlSanitizer like
# any client-submitted content, because a markdown document may embed raw HTML.
Markdown==3.10.3
alembic==1.18.4
